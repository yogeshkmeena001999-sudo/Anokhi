const express = require('http');
const http = require('http');
const { Server } = require('socket.io');
const { Pool } = require('pg');
const crypto = require('crypto');
require('dotenv').config();

const app = express();
const server = http.createServer(app);
const io = new Server(server, {
  cors: { origin: "*", methods: ["GET", "POST", "PATCH"] }
});

// PostgreSQL Connection Pool
const pool = new Pool({
  connectionString: process.env.DATABASE_URL,
  ssl: process.env.NODE_ENV === 'production' ? { rejectUnauthorized: false } : false
});

app.use(express.json());

// --- 1. SERVICEABILITY & GEOFENCING API ---
app.get('/api/v1/stores/serviceability', async (req, res) => {
  const { lat, lng } = req.query;
  if (!lat || !lng) return res.status(400).json({ error: 'Latitude and longitude are required' });

  try {
    // PostGIS query to find nearest active dark store within its delivery radius
    const query = `
      id, name, address, delivery_radius_meters,
      ST_Distance(location, ST_SetSRID(ST_MakePoint($1, $2), 4326)::geography) AS distance_meters
      FROM dark_stores
      WHERE is_active = TRUE
      ORDER BY distance_meters ASC
      LIMIT 1;
    `;
    const result = await pool.query(query, [parseFloat(lng), parseFloat(lat)]);
    
    if (result.rows.length === 0) {
      return res.status(404).json({ serviceable: false, message: 'No dark stores nearby.' });
    }

    const store = result.rows[0];
    if (store.distance_meters > store.delivery_radius_meters) {
      return res.json({ serviceable: false, message: 'Outside quick-delivery radius (max 3km).' });
    }

    res.json({ serviceable: true, storeId: store.id, storeName: store.name, distanceKm: (store.distance_meters / 1000).toFixed(2), etaMinutes: 8 });
  } catch (err) {
    res.status(500).json({ error: err.message });
  }
});

// --- 2. CHECKOUT & INVENTORY LOCKING API ---
app.post('/api/v1/orders/checkout', async (req, res) => {
  const { customerId, storeId, items, deliveryAddress, deliveryLat, deliveryLng, paymentMethod, subtotal, grandTotal, tipAmount } = req.body;
  const client = await pool.connect();

  try {
    await client.query('BEGIN');

    // Atomic inventory lock check
    for (let item of items) {
      const stockCheck = await client.query(
        'SELECT available_stock FROM store_inventory WHERE store_id = $1 AND product_id = $2 FOR UPDATE',
        [storeId, item.productId]
      );
      if (stockCheck.rows.length === 0 || stockCheck.rows[0].available_stock < item.quantity) {
        throw new Error(`Product ${item.productId} is out of stock.`);
      }
    }

    const handoverOtp = Math.floor(1000 + Math.random() * 9000).toString();
    const orderNumber = 'BLK-' + Math.floor(10000 + Math.random() * 90000);

    const orderQuery = `
      INSERT INTO orders (order_number, customer_id, store_id, delivery_address, delivery_location, status, payment_method, subtotal, grand_total, tip_amount, handover_otp)
      VALUES ($1, $2, $3, $4, ST_SetSRID(ST_MakePoint($5, $6), 4326)::geography, 'PLACED', $7, $8, $9, $10, $11)
      RETURNING id, order_number;
    `;
    const orderResult = await client.query(orderQuery, [
      orderNumber, customerId, storeId, deliveryAddress, deliveryLng, deliveryLat, paymentMethod, subtotal, grandTotal, tipAmount, handoverOtp
    ]);
    const orderId = orderResult.rows[0].id;

    // Deduct stock
    for (let item of items) {
      await client.query(
        'UPDATE store_inventory SET available_stock = available_stock - $1 WHERE store_id = $2 AND product_id = $3',
        [item.quantity, storeId, item.productId]
      );
      await client.query(
        'INSERT INTO order_items (order_id, product_id, quantity, unit_price, gst_rate) VALUES ($1, $2, $3, $4, 5.00)',
        [orderId, item.productId, item.quantity, item.unitPrice]
      );
    }

    await client.query('COMMIT');

    // Broadcast new order to Dark Store tablet room via Socket.io
    io.to(`store:${storeId}`).emit('store:new_order', { orderId, orderNumber, grandTotal, items });

    res.status(201).json({ success: true, orderId, orderNumber, handoverOtp });
  } catch (err) {
    await client.query('ROLLBACK');
    res.status(400).json({ error: err.message });
  } finally {
    client.release();
  }
});

// --- 3. RAZORPAY PAYMENT WEBHOOK VERIFICATION ---
app.post('/api/v1/webhooks/razorpay', async (req, res) => {
  const secret = process.env.RAZORPAY_WEBHOOK_SECRET;
  const shasum = crypto.createHmac('sha256', secret);
  shasum.update(JSON.stringify(req.body));
  const digest = shasum.digest('hex');

  if (digest === req.headers['x-razorpay-signature']) {
    const event = req.body.event;
    if (event === 'payment.captured') {
      const paymentEntity = req.body.payload.payment.entity;
      const orderId = paymentEntity.notes.internal_order_id;
      
      await pool.query("UPDATE orders SET payment_status = 'PAID' WHERE id = $1", [orderId]);
    }
    res.json({ status: 'ok' });
  } else {
    res.status(400).json({ error: 'Invalid signature' });
  }
});

// --- 4. REAL-TIME SOCKET.IO GPS & RIDER TRACKING ---
io.on('connection', (socket) => {
  console.log('Client connected:', socket.id);

  socket.on('join:store_room', (storeId) => socket.join(`store:${storeId}`));
  socket.on('join:order_room', (orderId) => socket.join(`order:${orderId}`));

  // Rider transmits live GPS coordinates
  socket.on('rider:location_update', (data) => {
    const { orderId, lat, lng, heading } = data;
    // Broadcast rider coordinates directly to customer tracking room
    io.to(`order:${orderId}`).emit('order:rider_moved', { lat, lng, heading });
  });

  socket.on('disconnect', () => console.log('Client disconnected:', socket.id));
});

const PORT = process.env.PORT || 5000;
server.listen(PORT, () => console.log(`Quick-commerce production backend running on port ${PORT}`));
